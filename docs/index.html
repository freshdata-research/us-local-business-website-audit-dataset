Python Quickstart

Inspect HTTP defect distributions directly using standard library Python:

import csv, urllib.request, io
from collections import Counter

url = "https://freshdata.today/samples/freshdata-texas-b2b-website-audit-sample.csv"
text = urllib.request.urlopen(url).read().decode("utf-8")
rows = list(csv.DictReader(io.StringIO(text)))
print("Total sample rows:", len(rows))
print("Defect status counts:", dict(Counter(r["Website_Status"] for r in rows)))
